-- =========================
-- TOOL DETECTION
-- =========================
local function getPan()
    local char = player.Character or player.CharacterAdded:Wait()
    for _, v in pairs(char:GetChildren()) do
        if v:IsA("Tool") and string.find(v.Name:lower(), "pan") then
            return v
        end
    end
    return nil
end

-- =========================
-- PROSPECT ACTION
-- =========================
local function prospectOnce()
    local pan = getPan()
    if not pan then return end

    local scripts = pan:FindFirstChild("Scripts")
    if not scripts then return end

    local collect   = scripts:FindFirstChild("Collect")
    local shake     = scripts:FindFirstChild("Shake")
    local panRemote = scripts:FindFirstChild("Pan")

    if collect and collect:IsA("RemoteFunction") then
        pcall(function() collect:InvokeServer(1, true) end)
    end
    if panRemote and panRemote:IsA("RemoteFunction") then
        pcall(function() panRemote:InvokeServer() end)
    end
    if shake and shake:IsA("RemoteEvent") then
        pcall(function() shake:FireServer() end)
    end
end

-- Auto prospect background loop
task.spawn(function()
    while true do
        if AutoProspect then
            prospectOnce()
            task.wait(Delay)
        else
            task.wait(0.25)
        end
    end
end)

-- =========================
-- TELEPORT HELPERS
-- =========================
local function findPlayer(name)
    local lname = name:lower()
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():find(lname, 1, true) then
            return p
        end
    end
    return nil
end

local function doTeleport()
    if teleportTarget == "" then
        Mixusfield:Notify({
            Title   = "Teleport",
            Content = "No target set. Enter a username first.",
            Type    = "warning",
        })
        return false
    end

    local target = findPlayer(teleportTarget)
    if not target then
        Mixusfield:Notify({
            Title   = "Teleport Failed",
            Content = "Player '" .. teleportTarget .. "' not found in server.",
            Type    = "error",
        })
        return false
    end

    if target == player then
        Mixusfield:Notify({
            Title   = "Teleport",
            Content = "You cannot teleport to yourself.",
            Type    = "warning",
        })
        return false
    end

    local targetChar = target.Character
    local localChar  = player.Character

    if not targetChar or not targetChar:FindFirstChild("HumanoidRootPart") then
        Mixusfield:Notify({
            Title   = "Teleport Failed",
            Content = target.Name .. "'s character is not available.",
            Type    = "error",
        })
        return false
    end

    if not localChar or not localChar:FindFirstChild("HumanoidRootPart") then
        Mixusfield:Notify({
            Title   = "Teleport Failed",
            Content = "Your character is not loaded.",
            Type    = "error",
        })
        return false
    end

    localChar.HumanoidRootPart.CFrame =
        targetChar.HumanoidRootPart.CFrame * CFrame.new(2, 0, 0)
    return true
end

-- =========================
-- FORMAT PRICE
-- =========================
local function formatPrice(n)
    if n == 0 then return "Free" end
    if n >= 1000000000000 then return string.format("$%.1fT", n / 1000000000000)
    elseif n >= 1000000000 then return string.format("$%.1fB", n / 1000000000)
    elseif n >= 1000000    then return string.format("$%.1fM", n / 1000000)
    elseif n >= 1000       then return string.format("$%.1fK", n / 1000)
    else return "$" .. n end
end

-- =========================
-- BUY FUNCTION
-- =========================
local function buyItem(itemName, itemData)
    local remotes = game:GetService("ReplicatedStorage"):FindFirstChild("Remotes")
        or game:GetService("ReplicatedStorage"):FindFirstChild("Remote")
        or game:GetService("ReplicatedStorage")

    local buyRemote = remotes and (
        remotes:FindFirstChild("BuyItem")  or
        remotes:FindFirstChild("Purchase") or
        remotes:FindFirstChild("Buy")      or
        remotes:FindFirstChild("ShopBuy")
    )

    if buyRemote and buyRemote:IsA("RemoteFunction") then
        local ok, result = pcall(function()
            return buyRemote:InvokeServer(itemName)
        end)
        if ok then
            Mixusfield:Notify({ Title = "Purchased!", Content = "Bought: " .. itemName, Type = "success" })
        else
            Mixusfield:Notify({ Title = "Purchase Failed", Content = tostring(result), Type = "error" })
        end
    elseif buyRemote and buyRemote:IsA("RemoteEvent") then
        pcall(function() buyRemote:FireServer(itemName) end)
        Mixusfield:Notify({ Title = "Purchased!", Content = "Bought: " .. itemName, Type = "success" })
    else
        Mixusfield:Notify({
            Title   = itemName,
            Content = "Price: " .. formatPrice(itemData.Price) ..
                (itemData.Effect and (" | " .. itemData.Effect) or ""),
            Type = "info",
        })
    end
end

    Name     = "Enable Teleport Loop",
    Default  = false,
    Icon     = { Type = "Lucide", Source = "repeat" },
    Callback = function(enabled)
        teleportLoopActive = enabled

        if enabled then
            -- guard: need a target first
            if teleportTarget == "" then
                teleportLoopActive = false
                Mixusfield:Notify({
                    Title   = "Loop Teleport",
                    Content = "Set a target username first!",
                    Type    = "warning",
                })
                return
            end

            Mixusfield:Notify({
                Title   = "Loop Teleport ON",
                Content = "Looping to " .. teleportTarget .. " every " .. teleportDelay .. "s.",
                Type    = "success",
            })

            -- spawn the loop thread
            teleportLoopThread = task.spawn(function()
                while teleportLoopActive do
                    doTeleport()
                    task.wait(teleportDelay)
                end
            end)

        else
            -- kill thread immediately — no waiting for next tick
            if teleportLoopThread then
                task.cancel(teleportLoopThread)
                teleportLoopThread = nil
            end
            teleportLoopActive = false
            Mixusfield:Notify({
                Title   = "Loop Teleport OFF",
                Content = "Teleport loop stopped.",
                Type    = "warning",
            })
        end
    end,
})
