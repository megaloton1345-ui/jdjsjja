local TeleportTimer    = 0.3     -- seconds to stay at forward position
local ReturnDelay      = 0.2     -- seconds to wait after returning before next cycle
local CompleteAllRunning = false

-- =====================================================
-- TOOL DETECTION
-- =====================================================
local function getPan()
    local char = player.Character or player.CharacterAdded:Wait()
    for _, v in pairs(char:GetChildren()) do
        if v:IsA("Tool") and string.find(v.Name:lower(), "pan") then
            return v
        end
    end
    return nil
end

-- =====================================================
-- TELEPORT HELPERS
-- =====================================================
local function getHRP()
    local char = player.Character
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart")
end

local function teleportTo(cf)
    local hrp = getHRP()
    if hrp then
        hrp.CFrame = cf
    end
end

-- Moves forward by `studs` on X/Z only, preserves Y and rotation
local function getFrontPosition(studs)
    local hrp = getHRP()
    if not hrp then return nil end
    local cf   = hrp.CFrame
    local look = cf.LookVector
    local flat = Vector3.new(look.X, 0, look.Z).Unit
    local newPos = cf.Position + flat * studs
    return CFrame.new(newPos, newPos + flat)
end

-- =====================================================
-- PROSPECT ACTION  (with configurable teleport timer)
-- =====================================================
local function prospectOnce()
    local pan = getPan()
    if not pan then return end

    local scripts   = pan:FindFirstChild("Scripts")
    if not scripts then return end

    local collect   = scripts:FindFirstChild("Collect")
    local shake     = scripts:FindFirstChild("Shake")
    local panRemote = scripts:FindFirstChild("Pan")

    local hrp        = getHRP()
    local originalCF = hrp and hrp.CFrame or nil

    -- Teleport forward
    if originalCF then
        local frontCF = getFrontPosition(TeleportStuds)
        if frontCF then
            teleportTo(frontCF)
            task.wait(TeleportTimer)   -- ← configurable timer
        end
    end

    -- Fire remotes
    if collect and collect:IsA("RemoteFunction") then
        pcall(function() collect:InvokeServer(1, true) end)
    end
    if panRemote and panRemote:IsA("RemoteFunction") then
        pcall(function() panRemote:InvokeServer() end)
    end
    if shake and shake:IsA("RemoteEvent") then
        pcall(function() shake:FireServer() end)
    end

    -- Return to original position
    if originalCF then
        task.wait(ReturnDelay)         -- ← configurable return delay slider
        teleportTo(originalCF)
    end
end

-- =====================================================
-- SELL ALL ACTION
-- =====================================================
local function sellAll()
    pcall(function()
        ReplicatedStorage
            :WaitForChild("Remotes")
            :WaitForChild("Shop")
            :WaitForChild("SellAll")
            :InvokeServer()
    end)
end

-- =====================================================
-- AUTO LOOPS  (both respect MasterRun)
-- =====================================================
task.spawn(function()
    while true do
        if MasterRun and AutoProspect then
            prospectOnce()
            task.wait(Delay)
        else
            task.wait(0.25)
        end
    end
end)

task.spawn(function()
    while true do
        if MasterRun and AutoSell then
            sellAll()
            task.wait(3)
        else
            task.wait(0.5)
        end
    end
end)

-- =====================================================
-- COMPLETE ALL  (one-shot: prospect + sell)
-- =====================================================
local function completeAll()
    if CompleteAllRunning then return end
    CompleteAllRunning = true
    prospectOnce()
    task.wait(0.5)
    sellAll()
    CompleteAllRunning = false
end

-- =====================================================
-- FORMAT PRICE HELPER
-- =====================================================
local function formatPrice(n)
    if n == 0 then return "Free" end
    if n >= 1000000000000 then return string.format("$%.1fT", n / 1000000000000)
    elseif n >= 1000000000 then return string.format("$%.1fB", n / 1000000000)
    elseif n >= 1000000    then return string.format("$%.1fM", n / 1000000)
    elseif n >= 1000       then return string.format("$%.1fK", n / 1000)
    else return "$" .. n end
end

-- =====================================================
-- BUY FUNCTION
-- =====================================================
local function buyItem(itemName, itemData)
    local success, result = pcall(function()
        local shopItem = workspace
            :WaitForChild("Purchasable")
            :WaitForChild(itemData.Zone)
            :WaitForChild(itemName)
            :WaitForChild("ShopItem")
        ReplicatedStorage
            :WaitForChild("Remotes")
            :WaitForChild("Shop")
            :WaitForChild("BuyItem")
            :InvokeServer(shopItem, 1)
    end)
    if success then
        Mixusfield:Notify({ Title = "Purchased!", Content = itemName .. " bought from " .. itemData.Zone, Type = "success" })
    else
        Mixusfield:Notify({ Title = "Purchase Failed", Content = tostring(result), Type = "error" })
    end
end
