local TeleportTab = Window:CreateTab({
    Name = "Teleport",
    Icon = { Type = "Lucide", Source = "map-pin" },

-- ── Teleport State ────────────────────────────────────────
local teleportLoopActive = false
local teleportLoopThread = nil
local teleportTarget      = ""
local teleportDelay       = 3  -- seconds between each teleport

-- Helper: find a player by (partial) name
local function findPlayer(name)
    local lname = name:lower()
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p.Name:lower():find(lname, 1, true) then
            return p
        end
    end
    return nil
end

-- Helper: teleport local player to a target player
local function doTeleport()
    local localPlayer = game.Players.LocalPlayer
    local target = findPlayer(teleportTarget)

    if teleportTarget == "" then
        Mixusfield:Notify({
            Title   = "Teleport",
            Content = "No target set. Enter a username first.",
            Type    = "warning",
        })
        return false
    end

    if not target then
        Mixusfield:Notify({
            Title   = "Teleport Failed",
            Content = "Player '" .. teleportTarget .. "' not found.",
            Type    = "error",
        })
        return false
    end

    if target == localPlayer then
        Mixusfield:Notify({
            Title   = "Teleport",
            Content = "You cannot teleport to yourself!",
            Type    = "warning",
        })
        return false
    end

    local targetChar = target.Character
    local localChar  = localPlayer.Character

    if not targetChar or not targetChar:FindFirstChild("HumanoidRootPart") then
        Mixusfield:Notify({
            Title   = "Teleport Failed",
            Content = target.Name .. " has no character.",
            Type    = "error",
        })
        return false
    end

    if not localChar or not localChar:FindFirstChild("HumanoidRootPart") then
        Mixusfield:Notify({
            Title   = "Teleport Failed",
            Content = "Your character is not loaded.",
            Type    = "error",
        })
        return false
    end

    localChar.HumanoidRootPart.CFrame = targetChar.HumanoidRootPart.CFrame + Vector3.new(2, 0, 0)
    return true
end

TeleportTab:CreateSection({ Name = "Loop Teleport" })

-- Loop toggle — ON keeps looping, OFF stops the loop
local LoopTeleportToggle = TeleportTab:CreateToggle({
    Name    = "Loop Teleport",
    Default = false,
    Icon    = { Type = "Lucide", Source = "repeat" },
    Callback = function(enabled)
        teleportLoopActive = enabled

        if enabled then
            -- Start the loop in a new thread
            Mixusfield:Notify({
                Title   = "Loop Teleport",
                Content = "Loop started — teleporting every " .. teleportDelay .. "s.",
                Type    = "info",
            })

            teleportLoopThread = task.spawn(function()
                while teleportLoopActive do
                    doTeleport()
                    task.wait(teleportDelay)
                end
            end)
        else
            -- Stop the loop
            teleportLoopActive = false
            if teleportLoopThread then
                task.cancel(teleportLoopThread)
                teleportLoopThread = nil
            end
            Mixusfield:Notify({
                Title   = "Loop Teleport",
                Content = "Loop stopped.",
                Type    = "warning",
            })
        end
    end,
})
