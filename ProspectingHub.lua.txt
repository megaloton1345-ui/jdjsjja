-- =====================================================
-- STATE
-- =====================================================
local AutoProspect  = false

local TeleportStuds = 10     -- 1 – 50 studs

-- =====================================================
-- CHARACTER HELPERS
-- =====================================================
local function getHRP()
    local char = player.Character
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart")
end

local function getHumanoid()
    local char = player.Character
    if not char then return nil end
    return char:FindFirstChildOfClass("Humanoid")
end

local function getPan()
    local char = player.Character
    if not char then return nil end
    for _, v in pairs(char:GetChildren()) do
        if v:IsA("Tool") and v.Name:lower():find("pan") then
            return v
        end
    end
    return nil
end

-- =====================================================
-- CORE PROSPECT ACTION
-- Fires Collect + Pan + Shake directly — no teleport needed
-- =====================================================
local function prospectOnce()
    local pan = getPan()
    if not pan then return end

    local scripts = pan:FindFirstChild("Scripts")
    if not scripts then return end

    local collect   = scripts:FindFirstChild("Collect")
    local shake     = scripts:FindFirstChild("Shake")
    local panRemote = scripts:FindFirstChild("Pan")

    if collect and collect:IsA("RemoteFunction") then
        pcall(function() collect:InvokeServer(1, true) end)
    end

    if panRemote and panRemote:IsA("RemoteFunction") then
        pcall(function() panRemote:InvokeServer() end)
    end

    if shake and shake:IsA("RemoteEvent") then
        pcall(function() shake:FireServer() end)
    end
end

-- =====================================================
-- SELL ACTION
-- =====================================================
local function sellAll()
    pcall(function()
        ReplicatedStorage
            :WaitForChild("Remotes")
            :WaitForChild("Shop")
            :WaitForChild("SellAll")
            :InvokeServer()
    end)
end

-- =====================================================
-- TELEPORT ACTION (independent, only used in Teleport tab)
-- =====================================================
local function getFrontPosition(studs)
    local hrp = getHRP()
    if not hrp then return nil end
    local look = hrp.CFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z).Unit
    local newPos = hrp.CFrame.Position + flat * studs
    return CFrame.new(newPos, newPos + flat)
end

local function teleportTo(cf)
    local hrp = getHRP()
    if hrp then hrp.CFrame = cf end
end

-- =====================================================
-- AUTO LOOPS
-- =====================================================
task.spawn(function()
    while true do
        if AutoProspect then
            prospectOnce()
            task.wait(ProspectDelay)
        else
            task.wait(0.2)
        end
    end
end)

task.spawn(function()
    while true do
        if AutoSell then
            sellAll()
            task.wait(3)
        else
            task.wait(0.5)
        end
    end
end)

-- =====================================================
-- COMPLETE ALL  (prospect + sell in one cycle)
-- =====================================================
local function completeAll()
    prospectOnce()
    task.wait(0.5)
    sellAll()
end

local function formatPrice(n)
    if n == 0 then return "Free" end
    if n >= 1000000000000 then return string.format("$%.1fT", n/1000000000000)
    elseif n >= 1000000000 then return string.format("$%.1fB", n/1000000000)
    elseif n >= 1000000    then return string.format("$%.1fM", n/1000000)
    elseif n >= 1000       then return string.format("$%.1fK", n/1000)
    else return "$"..n end
end

local function buyItem(itemName, itemData)
    local ok, err = pcall(function()
        local shopItem = workspace
            :WaitForChild("Purchasable")
            :WaitForChild(itemData.Zone)
            :WaitForChild(itemName)
            :WaitForChild("ShopItem")
        ReplicatedStorage
            :WaitForChild("Remotes")
            :WaitForChild("Shop")
            :WaitForChild("BuyItem")
            :InvokeServer(shopItem, 1)
    end)
    if ok then
        Mixusfield:Notify({ Title = "Purchased!", Content = itemName.." bought from "..itemData.Zone, Type = "success" })
    else
        Mixusfield:Notify({ Title = "Purchase Failed", Content = tostring(err), Type = "error" })
    end
end

    Callback = function()

        if not hrp then
            Mixusfield:Notify({ Title = "Error", Content = "Character not found.", Type = "error" })
            return
        end
        local originalCF = hrp.CFrame
        local frontCF    = getFrontPosition(TeleportStuds)
        if frontCF then
            teleportTo(frontCF)
            Mixusfield:Notify({ Title = "Teleported", Content = "Moved "..TeleportStuds.." studs. Returning in "..TeleportTimer.."s…", Type = "info" })
            task.wait(TeleportTimer)
            teleportTo(originalCF)
            Mixusfield:Notify({ Title = "Returned", Content = "Back to original spot.", Type = "success" })
        end
    end,
})
