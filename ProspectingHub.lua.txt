local ProspectDelay = 0.5         -- lowered default from 1 → 0.5

-- ── CACHED REMOTES (fetched once, reused) ────────────
local _remotes       = nil
local _shopRemote    = nil

local function getRemotes()
    if _remotes then return _remotes end
    local ok, result = pcall(function()
        return ReplicatedStorage:WaitForChild("Remotes", 5)
    end)
    if ok and result then
        _remotes = result
        return _remotes
    end
    return nil
end

local function getShopRemote(name)
    local remotes = getRemotes()
    if not remotes then return nil end
    local ok, result = pcall(function()
        return remotes:WaitForChild("Shop", 5):WaitForChild(name, 5)
    end)
    return ok and result or nil
end

-- ── CHARACTER HELPERS ────────────────────────────────
local function getHRP()
    local char = player.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function getHumanoid()
    local char = player.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end

local function getPan()
    local char = player.Character
    if not char then return nil end
    for _, v in ipairs(char:GetChildren()) do
        if v:IsA("Tool") and v.Name:lower():find("pan") then
            return v
        end
    end
    return nil
end

-- ── CACHED PAN SCRIPTS ───────────────────────────────
-- Re-cache whenever pan changes for speed
local _cachedPan     = nil
local _collect       = nil
local _panRemote     = nil
local _shake         = nil

local function refreshPanCache()
    local pan = getPan()
    if pan == _cachedPan then return end   -- nothing changed
    _cachedPan  = pan
    _collect    = nil
    _panRemote  = nil
    _shake      = nil
    if not pan then return end
    local scripts = pan:FindFirstChild("Scripts")
    if not scripts then return end
    _collect   = scripts:FindFirstChild("Collect")
    _panRemote = scripts:FindFirstChild("Pan")
    _shake     = scripts:FindFirstChild("Shake")
end

-- ── CORE PROSPECT ────────────────────────────────────
-- Fires all three remotes with no unnecessary waits
local function prospectOnce()
    refreshPanCache()
    if not _cachedPan then return false end

    if _collect and _collect:IsA("RemoteFunction") then
        pcall(_collect.InvokeServer, _collect, 1, true)
    end
    if _panRemote and _panRemote:IsA("RemoteFunction") then
        pcall(_panRemote.InvokeServer, _panRemote)
    end
    if _shake and _shake:IsA("RemoteEvent") then
        pcall(_shake.FireServer, _shake)
    end
    return true
end

-- Burst: fires N times back-to-back with minimal spacing
local function prospectBurst()
    for i = 1, BurstCount do
        prospectOnce()
        if i < BurstCount then
            task.wait(0.05)   -- tiny gap between bursts to avoid throttle
        end
    end
end

-- ── SELL ─────────────────────────────────────────────
local _sellRemote = nil

local function sellAll()
    if not _sellRemote then
        _sellRemote = getShopRemote("SellAll")
    end
    if _sellRemote then
        pcall(_sellRemote.InvokeServer, _sellRemote)
    end
end

-- ── COMPLETE ALL ─────────────────────────────────────
local function completeAll()
    local fired = BurstMode and prospectBurst() or prospectOnce()
    if fired then
        task.wait(0.3)    -- brief settle before selling
        sellAll()
    end
end

-- ── TELEPORT HELPERS ─────────────────────────────────
local function getFrontPosition(studs)
    local hrp = getHRP()
    if not hrp then return nil end
    local look = hrp.CFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z).Unit
    local newPos = hrp.CFrame.Position + flat * studs
    return CFrame.new(newPos, newPos + flat)
end

local function teleportTo(cf)
    local hrp = getHRP()
    if hrp then hrp.CFrame = cf end
end

-- ── AUTO LOOPS ───────────────────────────────────────
-- Prospect loop: efficient, no idle spinning
task.spawn(function()
    while true do
        if AutoProspect then
            if BurstMode then
                prospectBurst()
            else
                prospectOnce()
            end
            task.wait(math.max(ProspectDelay, 0.05))
        else
            task.wait(0.1)
        end
    end
end)

-- Sell loop
task.spawn(function()
    while true do
        if AutoSell then
            sellAll()
            task.wait(SellInterval)
        else
            task.wait(0.5)
        end
    end
end)

-- ── HELPERS ──────────────────────────────────────────
local function formatPrice(n)
    if n == 0 then return "Free" end
    if     n >= 1e12 then return string.format("$%.1fT", n / 1e12)
    elseif n >= 1e9  then return string.format("$%.1fB", n / 1e9)
    elseif n >= 1e6  then return string.format("$%.1fM", n / 1e6)
    elseif n >= 1e3  then return string.format("$%.1fK", n / 1e3)
    else                  return "$" .. n
    end
end

local function sortedKeys(tbl, pricePath)
    local keys = {}
    for k in pairs(tbl) do keys[#keys + 1] = k end
    table.sort(keys, function(a, b)
        return (pricePath(tbl[a])) < (pricePath(tbl[b]))
    end)
    return keys
end

local getPrice = function(d) return d.Price end

local function buyItem(itemName, itemData)
    local ok, err = pcall(function()
        local shopItem = workspace
            :WaitForChild("Purchasable", 5)
            :WaitForChild(itemData.Zone,  5)
            :WaitForChild(itemName,       5)
            :WaitForChild("ShopItem",     5)
        local buyRemote = getShopRemote("BuyItem")
        if not buyRemote then error("BuyItem remote not found") end
        buyRemote:InvokeServer(shopItem, 1)
    end)
    if ok then
        Mixusfield:Notify({
            Title   = "Purchased!",
            Content = itemName .. " bought from " .. itemData.Zone,
            Type    = "success",
        })
    else
        Mixusfield:Notify({
            Title   = "Purchase Failed",
            Content = tostring(err),
            Type    = "error",
        })
    end
end

local function buildPriceList(category, list, extraFn)
    local t = {}
    for _, n in ipairs(list) do
        local d    = category[n]
        local line = n .. " → " .. formatPrice(d.Price) .. " [" .. d.Zone .. "]"
        if extraFn then line = line .. extraFn(d) end
        t[#t + 1] = line
    end
    return table.concat(t, "\n")
end
